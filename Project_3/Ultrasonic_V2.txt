// 4 Ultrasonic sensor 13/08/2025
#include <stdint.h>

/* ------------------- Base addresses ------------------- */
#define RCC_BASE        0x40023800
#define GPIOA_BASE      0x40020000
#define GPIOB_BASE      0x40020400
#define GPIOC_BASE      0x40020800
#define TIM2_BASE       0x40000000

/* ------------------- RCC registers ------------------- */
#define RCC_AHB1ENR     (*(volatile uint32_t *)(RCC_BASE + 0x30))
#define RCC_APB1ENR     (*(volatile uint32_t *)(RCC_BASE + 0x40))

/* ------------------- GPIO registers ------------------- */
#define GPIO_MODER(port)   (*(volatile uint32_t *)((port) + 0x00))
#define GPIO_IDR(port)     (*(volatile uint32_t *)((port) + 0x10))
#define GPIO_ODR(port)     (*(volatile uint32_t *)((port) + 0x14))

/* ------------------- Timer registers ------------------- */
#define TIM2_CR1        (*(volatile uint32_t *)(TIM2_BASE + 0x00))
#define TIM2_CNT        (*(volatile uint32_t *)(TIM2_BASE + 0x24))
#define TIM2_PSC        (*(volatile uint32_t *)(TIM2_BASE + 0x28))
#define TIM2_ARR        (*(volatile uint32_t *)(TIM2_BASE + 0x2C))

/* ------------------- Port addresses ------------------- */
#define PORTA GPIOA_BASE
#define PORTB GPIOB_BASE
#define PORTC GPIOC_BASE

/* ------------------- Sensor pin mapping -------------------
   Sensor 1: TRIG=PA5, ECHO=PA6
   Sensor 2: TRIG=PA7, ECHO=PC0
   Sensor 3: TRIG=PC1, ECHO=PC2
   Sensor 4: TRIG=PC3, ECHO=PA4
------------------------------------------------------------*/
typedef struct {
    uint32_t trig_port;
    uint8_t  trig_pin;
    uint32_t echo_port;
    uint8_t  echo_pin;
} HCSR04_t;

HCSR04_t sensors[4] = {
    {PORTA, 5, PORTA, 6},
    {PORTA, 7, PORTC, 0},
    {PORTC, 1, PORTC, 2},
    {PORTC, 3, PORTA, 4}
};
uint32_t distances[4];
/* ------------------- Delay in microseconds ------------------- */
void delay_us(uint32_t us) {
    TIM2_CNT = 0;
    while (TIM2_CNT < us);
}

/* ------------------- Init GPIO ------------------- */
void gpio_init(void) {
    // Enable GPIOA, GPIOC clocks
    RCC_AHB1ENR |= (1 << 0) | (1 << 2);

    for (int i = 0; i < 4; i++) {
        // TRIG as output
        GPIO_MODER(sensors[i].trig_port) &= ~(3 << (sensors[i].trig_pin * 2));
        GPIO_MODER(sensors[i].trig_port) |=  (1 << (sensors[i].trig_pin * 2));

        // ECHO as input
        GPIO_MODER(sensors[i].echo_port) &= ~(3 << (sensors[i].echo_pin * 2));
    }
}

/* ------------------- Init TIM2 for 1 MHz ------------------- */
void tim2_init(void) {
    RCC_APB1ENR |= (1 << 0); // TIM2 clock enable
    TIM2_PSC = 16 - 1;       // 1 MHz -> 1 us tick (16 MHz HSI assumed)
    TIM2_ARR = 0xFFFF;
    TIM2_CR1 |= 1;           // Enable counter
}

/* ------------------- Read one ultrasonic sensor ------------------- */
uint32_t hcsr04_read(HCSR04_t *sensor) {
    uint32_t start_time, end_time;

    // Send 10us TRIG pulse
    GPIO_ODR(sensor->trig_port) &= ~(1 << sensor->trig_pin);
    delay_us(2);
    GPIO_ODR(sensor->trig_port) |= (1 << sensor->trig_pin);
    delay_us(10);
    GPIO_ODR(sensor->trig_port) &= ~(1 << sensor->trig_pin);

    // Wait for ECHO high
    while (!(GPIO_IDR(sensor->echo_port) & (1 << sensor->echo_pin)));

    start_time = TIM2_CNT;

    // Wait for ECHO low
    while (GPIO_IDR(sensor->echo_port) & (1 << sensor->echo_pin));

    end_time = TIM2_CNT;

    // Handle overflow
    uint32_t pulse_width = (end_time >= start_time) ?
                           (end_time - start_time) :
                           ((0xFFFF - start_time) + end_time);

    // Convert to cm
    return pulse_width / 58;
}

/* ------------------- Main ------------------- */
int main(void) {
    gpio_init();
    tim2_init();

    while (1) {
        for (int i = 0; i < 4; i++) {
            distances[i] = hcsr04_read(&sensors[i]);
            // Output distances[i] to UART/LCD if required
        }

        // crude delay between measurements
        for (volatile uint32_t i = 0; i < 500000; i++);
    }
}


//20/08/2025 4 Ultrasonic sensors with Traffic lights- Version 2

#include <stdint.h>

/* ---------------- RCC base ---------------- */
#define RCC_BASE        0x40023800
#define RCC_AHB1ENR     (*(volatile uint32_t *)(RCC_BASE + 0x30))
#define RCC_APB1ENR     (*(volatile uint32_t *)(RCC_BASE + 0x40))

/* ---------------- GPIO base ---------------- */
#define GPIOA_BASE      0x40020000
#define GPIOB_BASE      0x40020400
#define GPIOC_BASE      0x40020800

#define GPIO_MODER(port)   (*(volatile uint32_t *)((port) + 0x00))
#define GPIO_IDR(port)     (*(volatile uint32_t *)((port) + 0x10))
#define GPIO_ODR(port)     (*(volatile uint32_t *)((port) + 0x14))

/* ---------------- TIM2 ---------------- */
#define TIM2_BASE       0x40000000
#define TIM2_CR1        (*(volatile uint32_t *)(TIM2_BASE + 0x00))
#define TIM2_CNT        (*(volatile uint32_t *)(TIM2_BASE + 0x24))
#define TIM2_PSC        (*(volatile uint32_t *)(TIM2_BASE + 0x28))
#define TIM2_ARR        (*(volatile uint32_t *)(TIM2_BASE + 0x2C))

/* ---------------- Ports ---------------- */
#define PORTA GPIOA_BASE
#define PORTB GPIOB_BASE
#define PORTC GPIOC_BASE

/* ---------------- Ultrasonic struct ---------------- */
typedef struct {
    uint32_t trig_port;
    uint8_t  trig_pin;
    uint32_t echo_port;
    uint8_t  echo_pin;
} HCSR04_t;

HCSR04_t sensors[4] = {
    {PORTA, 5, PORTA, 6},   // Sensor 1
    {PORTA, 7, PORTC, 0},   // Sensor 2
    {PORTC, 1, PORTC, 2},   // Sensor 3
    {PORTC, 3, PORTA, 4}    // Sensor 4
};
uint32_t distances[4];

/* ---------------- Traffic Lights ---------------- */
typedef struct {
    uint32_t port;
    uint8_t red, yellow, green;
} Traffic_t;

Traffic_t lanes[4] = {
    {PORTA, 2, 3, 8},   // Lane 1
    {PORTA, 9, 10, 11}, // Lane 2
    {PORTB, 2, 3, 4},   // Lane 3
    {PORTB, 5, 6, 7}    // Lane 4
};

/* ---------------- Delay using TIM2 ---------------- */
void delay_us(uint32_t us) {
    TIM2_CNT = 0;
    while (TIM2_CNT < us);
}
void delay_ms(uint32_t ms) {
    while (ms--) delay_us(1000);
}

/* ---------------- Init GPIO ---------------- */
void gpio_init(void) {
    // Enable GPIOA, GPIOB, GPIOC
    RCC_AHB1ENR |= (1<<0) | (1<<1) | (1<<2);

    // Configure ultrasonic TRIG as output, ECHO as input
    for (int i = 0; i < 4; i++) {
        GPIO_MODER(sensors[i].trig_port) &= ~(3 << (sensors[i].trig_pin*2));
        GPIO_MODER(sensors[i].trig_port) |=  (1 << (sensors[i].trig_pin*2)); // output

        GPIO_MODER(sensors[i].echo_port) &= ~(3 << (sensors[i].echo_pin*2)); // input
    }

    // Configure traffic light pins as outputs
    for (int i = 0; i < 4; i++) {
        GPIO_MODER(lanes[i].port) &= ~(3 << (lanes[i].red*2));
        GPIO_MODER(lanes[i].port) |=  (1 << (lanes[i].red*2));
        GPIO_MODER(lanes[i].port) &= ~(3 << (lanes[i].yellow*2));
        GPIO_MODER(lanes[i].port) |=  (1 << (lanes[i].yellow*2));
        GPIO_MODER(lanes[i].port) &= ~(3 << (lanes[i].green*2));
        GPIO_MODER(lanes[i].port) |=  (1 << (lanes[i].green*2));
    }
}

/* ---------------- Init TIM2 ---------------- */
void tim2_init(void) {
    RCC_APB1ENR |= (1 << 0);  // TIM2 clock enable
    TIM2_PSC = 16 - 1;        // 1 MHz tick
    TIM2_ARR = 0xFFFF;
    TIM2_CR1 |= 1;
}

/* ---------------- Ultrasonic read ---------------- */
uint32_t hcsr04_read(HCSR04_t *sensor) {
    uint32_t start_time, end_time;

    // Send TRIG
    GPIO_ODR(sensor->trig_port) &= ~(1<<sensor->trig_pin);
    delay_us(2);
    GPIO_ODR(sensor->trig_port) |= (1<<sensor->trig_pin);
    delay_us(10);
    GPIO_ODR(sensor->trig_port) &= ~(1<<sensor->trig_pin);

    // Wait for ECHO high
    while (!(GPIO_IDR(sensor->echo_port) & (1<<sensor->echo_pin)));
    start_time = TIM2_CNT;

    // Wait for ECHO low
    while (GPIO_IDR(sensor->echo_port) & (1<<sensor->echo_pin));
    end_time = TIM2_CNT;

    uint32_t pulse = (end_time >= start_time) ? (end_time - start_time) : (0xFFFF - start_time + end_time);
    return pulse / 58;  // distance in cm
}

/* ---------------- Traffic control ---------------- */
void set_light(int lane, int r, int y, int g) {
    if (r) GPIO_ODR(lanes[lane].port) |= (1<<lanes[lane].red);
    else   GPIO_ODR(lanes[lane].port) &= ~(1<<lanes[lane].red);

    if (y) GPIO_ODR(lanes[lane].port) |= (1<<lanes[lane].yellow);
    else   GPIO_ODR(lanes[lane].port) &= ~(1<<lanes[lane].yellow);

    if (g) GPIO_ODR(lanes[lane].port) |= (1<<lanes[lane].green);
    else   GPIO_ODR(lanes[lane].port) &= ~(1<<lanes[lane].green);
}

void run_traffic_cycle(void) {
    for (int lane = 0; lane < 4; lane++) {
        // Green ON
        set_light(lane, 0, 0, 1);
        delay_ms(5000);

        // Yellow
        set_light(lane, 0, 1, 0);
        delay_ms(2000);

        // Red
        set_light(lane, 1, 0, 0);
    }
}

/* ---------------- Main ---------------- */
int main(void) {
    gpio_init();
    tim2_init();

    while (1) {
        // Read all ultrasonic sensors
        for (int i = 0; i < 4; i++) {
            distances[i] = hcsr04_read(&sensors[i]);
        }

        // Run traffic cycle
        run_traffic_cycle();
    }
}
